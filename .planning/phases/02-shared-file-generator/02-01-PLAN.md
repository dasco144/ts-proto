    Given options, returns a Code value whose toString() contains export declarations for all movable helpers.

    Cases:
    - generateSharedHelpers({ ...defaultOptions(), exportCommonSymbols: false }) -> output contains "export type DeepPartial" (GEN-03: forced export regardless of exportCommonSymbols)
    - generateSharedHelpers({ ...defaultOptions(), exportCommonSymbols: false }) -> output contains "export type Exact" (GEN-03)
    - generateSharedHelpers({ ...defaultOptions(), exportCommonSymbols: false }) -> output contains "export interface MessageFns" (GEN-03)
    - generateSharedHelpers({ ...defaultOptions(), globalThisPolyfill: true }) -> output.indexOf("const gt") < output.indexOf("DeepPartial") (GEN-04: gt before DeepPartial)
    - generateSharedHelpers(defaultOptions()) -> output does NOT contain "StructWrapperFns" (GEN-05)
    - generateSharedHelpers(defaultOptions()) -> output does NOT contain "AnyValueWrapperFns" (GEN-05)
    - generateSharedHelpers(defaultOptions()) -> output does NOT contain "fromProtoObjectId" (GEN-05)
    - generateSharedHelpers(defaultOptions()) -> output contains "export type Builtin" (GEN-02: basic helpers always present)
    - generateSharedHelpers(defaultOptions()) -> output contains "isSet" (GEN-02: basic comparison utils present)
    - generateSharedHelpers(defaultOptions()) -> output contains "isObject" (GEN-02)
  </behavior>
  <implementation>
    Create src/generate-shared-helpers.ts:

    ```typescript
    import { Code, ConditionalOutput, generateConditionalsUtils } from "ts-poet";
    import { makeUtils } from "./main";
    import { Options } from "./options";

    export function generateSharedHelpers(options: Options): Code {
      // GEN-03: Force exportCommonSymbols=true to ensure all helpers use `export`.
      // This only applies to the shared file — per-file generation uses the original options.
      const utils = makeUtils({ ...options, exportCommonSymbols: true });

      const helpers: ConditionalOutput[] = [];

      // GEN-04: globalThisPolyfill (the gt IIFE) MUST appear first.
      // utils.globalThis IS the IIFE object when globalThisPolyfill=true.
      // When globalThisPolyfill=false, utils.globalThis is an empty placeholder — do not include it.
      if (options.globalThisPolyfill) {
        helpers.push(utils.globalThis as ConditionalOutput);
      }

      // GEN-02: All movable helpers in order.
      // GEN-05: Excluded helpers are those that reference user-generated types:
      //   utils.StructWrapperFns, utils.AnyValueWrapperFns, utils.ListValueWrapperFns,
      //   utils.FieldMaskWrapperFns, utils.fromProtoObjectId, utils.fromJsonObjectId, utils.toProtoObjectId
      // utils.Long is an Import (not ConditionalOutput) and must never be in this array.
      helpers.push(
        utils.bytesFromBase64 as ConditionalOutput,
        utils.base64FromBytes as ConditionalOutput,
        utils.Builtin,
        utils.Exact,
        utils.DeepPartial,
        utils.numberToLong as ConditionalOutput,
        utils.longToNumber as ConditionalOutput,
        utils.toTimestamp as ConditionalOutput,
        utils.fromTimestamp as ConditionalOutput,
        utils.fromJsonTimestamp as ConditionalOutput,
        utils.isObject,
        utils.isSet,
        utils.MessageFns,
        utils.ExtensionFns,
        utils.ExtensionHolder,
        utils.NiceGrpcServerStreamingMethodResult as ConditionalOutput,
        utils.GrpcWebError as ConditionalOutput,
        utils.Extension,
        utils.fail as ConditionalOutput,
      );

      return generateConditionalsUtils(helpers);
    }
    ```

    PITFALL AVOIDANCE:
    - Do NOT use utils.Long in the helpers array (it is an Import, not ConditionalOutput)
    - Do NOT use helper.ifUsed — use generateConditionalsUtils which emits declarationSiteCode directly
    - Do NOT modify the makeUtils call in plugin.ts — only override exportCommonSymbols locally in generateSharedHelpers
    - When globalThisPolyfill=false, do NOT include utils.globalThis (it would be an empty placeholder that emits nothing but adds noise)
